-- Orb Drag + Click Open/Close Logic
local isMenuOpen = true
local function setMenuOpen(open)
    isMenuOpen = open
    MainFrame.Visible = open
    TweenService:Create(OrbRotor, TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
        Rotation = open and 0 or 90
    }):Play()
    OrbStroke.Color = open and ACCENT or Color3.fromRGB(120, 130, 150)
end

do
    local orbDragging = false
    local orbMoved = false
    local orbStartPos, orbStartMouse
    ToggleOrb.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            orbDragging = true
            orbMoved = false
            orbStartMouse = input.Position
            orbStartPos = ToggleOrb.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if orbDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - orbStartMouse
            if delta.Magnitude >= 6 then
                orbMoved = true
                ToggleOrb.Position = UDim2.new(
                    orbStartPos.X.Scale,
                    orbStartPos.X.Offset + delta.X,
                    orbStartPos.Y.Scale,
                    orbStartPos.Y.Offset + delta.Y
                )
            end
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if orbDragging and (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
            orbDragging = false
            if not orbMoved then
                setMenuOpen(not isMenuOpen)
            end
        end
    end)
end

local TopBar = Instance.new("Frame")

TopBar.BackgroundColor3 = Color3.fromRGB(16, 16, 22)
TopBar.BorderSizePixel = 0
TopBar.Parent = MainFrame

do
    local dragging, dragInput, dragStart, startPos
    TopBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = MainFrame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    TopBar.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            MainFrame.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)
end

local tabButtons = {}
local tabScrollFrames = {}
local langRefreshCallbacks = {}

local function registerLangRefresh(fn)
    table.insert(langRefreshCallbacks, fn)
end

local function switchTab(tabKey)
    activeTab = tabKey
    for k, sf in pairs(tabScrollFrames) do
        sf.Visible = (k == tabKey)
    end
    for k, btn in pairs(tabButtons) do
        if k == tabKey then
            btn.BackgroundTransparency = 0
            btn.BackgroundColor3 = ACCENT
            btn.TextColor3 = Color3.new(1, 1, 1)
        else
            btn.BackgroundTransparency = 1
            btn.TextColor3 = TEXT_MUTED
        end
    end
end

for _, key in ipairs(enabledTabs) do
    local btn = Instance.new("TextButton")
    btn.Name = "Tab_" .. key
    btn.Size = UDim2.new(1 / #enabledTabs, -3, 1, -4)
    btn.BackgroundColor3 = ACCENT

    btn.TextSize = 10
    btn.Font = Enum.Font.GothamBold
    btn.Parent = TabBar
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)

    local page = createScrollPage("Page_" .. key)
    tabButtons[key] = btn
    tabScrollFrames[key] = page

    btn.MouseButton1Click:Connect(function()
        switchTab(key)
    end)

    registerLangRefresh(function()
        btn.Text = T("tab_" .. key)
    end)
end

-- ESP Scanner System (100% Preserved from Spotlight_on_players)
local function removeESP(character)
    if espCache[character] then
        if espCache[character].highlight then espCache[character].highlight:Destroy() end
        if espCache[character].billboard then espCache[character].billboard:Destroy() end
        espCache[character] = nil
    end
end

local function addESP(character, modelName)
    if character == LocalPlayer.Character then return end
    if espCache[character] then return end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    local highlight = Instance.new("Highlight")
    highlight.Name = "ScannerHighlight"
    highlight.Adornee = character
    highlight.FillColor = Color3.fromRGB(255, 0, 0)
    highlight.FillTransparency = 0.4
    highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
    highlight.OutlineTransparency = 0
    highlight.Parent = character

    local billboard = Instance.new("BillboardGui")
    billboard.Name = "ScannerBillboard"
    billboard.Adornee = hrp
    billboard.Size = UDim2.new(0, 200, 0, 60)
    billboard.StudsOffset = Vector3.new(0, 3, 0)
    billboard.AlwaysOnTop = true
    billboard.Parent = character

    local textLabel = Instance.new("TextLabel")
    textLabel.Size = UDim2.new(1, 0, 1, 0)
    textLabel.BackgroundTransparency = 1
    textLabel.TextColor3 = Color3.fromRGB(255, 50, 50)
    textLabel.TextStrokeTransparency = 0
    textLabel.TextSize = 12
    textLabel.Font = Enum.Font.GothamBold
    textLabel.Parent = billboard

    espCache[character] = {
        highlight = highlight,
        billboard = billboard,
        textLabel = textLabel,
        rootPart = hrp,
        name = modelName
    }
end

ScanButton.MouseButton1Click:Connect(function()
    scanningEnabled = not scanningEnabled
    if scanningEnabled then
        ScanButton.BackgroundColor3 = Color3.fromRGB(150, 0, 0)
        setStatus(T("esp_on"))
        for _, player in ipairs(Players:GetPlayers()) do
            if player.Character then
                addESP(player.Character, player.Name)
            end
        end
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("Model") and obj:FindFirstChild("Humanoid") and obj:FindFirstChild("HumanoidRootPart") then
                local isPlayerChar = false
                for _, p in ipairs(Players:GetPlayers()) do
                    if p.Character == obj then isPlayerChar = true break end
                end
                if not isPlayerChar then
                    addESP(obj, obj.Name)
                end
            end
        end
    else
        ScanButton.BackgroundColor3 = BG_SURFACE
        setStatus(T("esp_off"))
        for char, _ in pairs(espCache) do
            removeESP(char)
        end
        espCache = {}
    end
end)

-- Teleport Front Helper (Exact CFrame from original)
local function teleportToCFrameFront(targetHRP)
    local myChar = LocalPlayer.Character
    if myChar and myChar:FindFirstChild("HumanoidRootPart") and targetHRP then
        myChar.HumanoidRootPart.CFrame = targetHRP.CFrame * CFrame.new(0, 0, -3.5) * CFrame.Angles(0, math.pi, 0)
    end
end

local function createToggleRow(parent, labelKey, initialState, onToggle)
    local row = Instance.new("Frame")
    row.Size = UDim2.new(1, -4, 0, 28)
    row.BackgroundColor3 = BG_SURFACE
    row.BorderSizePixel = 0
    row.Parent = parent
    Instance.new("UICorner", row).CornerRadius = UDim.new(0, 5)

    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(1, -54, 1, 0)
    lbl.Position = UDim2.new(0, 8, 0, 0)
    lbl.BackgroundTransparency = 1
    lbl.Text = T(labelKey)
    lbl.TextColor3 = TEXT_MAIN
    lbl.TextSize = 11
    lbl.Font = Enum.Font.GothamSemibold
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.TextTruncate = Enum.TextTruncate.AtEnd
    lbl.Parent = row

    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0, 40, 0, 20)
    btn.Position = UDim2.new(1, -44, 0.5, -10)
    btn.TextSize = 10
    btn.Font = Enum.Font.GothamBold
    btn.Parent = row
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)

    local state = initialState
    local function sync()
        btn.BackgroundColor3 = state and Color3.fromRGB(22, 163, 74) or BG_ELEVATED
        btn.TextColor3 = state and Color3.new(1, 1, 1) or TEXT_MUTED
        btn.Text = state and "ON" or "OFF"
    end
    sync()

    btn.MouseButton1Click:Connect(function()
        state = not state
        sync()
        onToggle(state)
    end)

    registerLangRefresh(function()
        lbl.Text = T(labelKey)
    end)

    return row
end

    plusBtn.TextColor3 = TEXT_MAIN
    plusBtn.TextSize = 13
    plusBtn.Font = Enum.Font.GothamBold

    Instance.new("UICorner", plusBtn).CornerRadius = UDim.new(0, 4)

    local resetStepBtn = Instance.new("TextButton")
    resetStepBtn.Size = UDim2.new(0, 22, 0, 20)
    resetStepBtn.Position = UDim2.new(1, -26, 0.5, -10)
    resetStepBtn.BackgroundColor3 = BG_ELEVATED
    resetStepBtn.Text = "↺"
    resetStepBtn.TextColor3 = ACCENT
    resetStepBtn.TextSize = 11
    resetStepBtn.Font = Enum.Font.GothamBold
    resetStepBtn.Parent = row
    Instance.new("UICorner", resetStepBtn).CornerRadius = UDim.new(0, 4)

    local val = initialVal
    local function update(nextVal)
        val = math.clamp(nextVal, minVal, maxVal)
        lbl.Text = T(labelKey) .. ": " .. tostring(val)
        onChange(val)
    end

    minusBtn.MouseButton1Click:Connect(function() update(val - step) end)
    plusBtn.MouseButton1Click:Connect(function() update(val + step) end)
    resetStepBtn.MouseButton1Click:Connect(function() update(initialVal) end)

    registerLangRefresh(function()
        lbl.Text = T(labelKey) .. ": " .. tostring(val)
    end)

    return row
end

local flyBV, flyBG
local function stopFly()
    if flyBV then flyBV:Destroy() flyBV = nil end
    if flyBG then flyBG:Destroy() flyBG = nil end
    local char = LocalPlayer.Character
    if char and char:FindFirstChildOfClass("Humanoid") then
        char:FindFirstChildOfClass("Humanoid").PlatformStand = false
    end
end

local function startFly()
    stopFly()
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end

    hum.PlatformStand = true
    flyBG = Instance.new("BodyGyro")
    flyBG.P = 9e4
    flyBG.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    flyBG.CFrame = hrp.CFrame
    flyBG.Parent = hrp

    flyBV = Instance.new("BodyVelocity")
    flyBV.Velocity = Vector3.zero
    flyBV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    flyBV.Parent = hrp
end

local function applyInvisibility(state)
    local char = LocalPlayer.Character
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
            part.Transparency = state and 1 or 0
        elseif part:IsA("Decal") then
            part.Transparency = state and 1 or 0
        end
    end
end

local function setCharacterHiddenLocal(char, hide)
    if not char then return end
    for _, obj in ipairs(char:GetDescendants()) do
        if obj:IsA("BasePart") and obj.Name ~= "HumanoidRootPart" then
            obj.LocalTransparencyModifier = hide and 1 or 0
        elseif obj:IsA("Decal") then
            obj.Transparency = hide and 1 or 0
        elseif obj:IsA("BillboardGui") then
            obj.Enabled = not hide
        end
    end
end

-- Main RenderStepped Loop (100% Original Spectate + Auto Bot Scan + ESP + Fly + Godmode)
RunService.RenderStepped:Connect(function()
    -- Camera Spectate (Players & Auto-Reacquiring Bots)
    if activeSpectatingPlayer then
        if activeSpectatingPlayer.Character and activeSpectatingPlayer.Character:FindFirstChild("Humanoid") then
            Camera.CameraSubject = activeSpectatingPlayer.Character.Humanoid
            local hrp = activeSpectatingPlayer.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                pcall(function() LocalPlayer:RequestStreamAroundAsync(hrp.Position) end)
            end
        else
            local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if hum then Camera.CameraSubject = hum end
        end
    elseif activeSpectatingModel then
        if not activeSpectatingModel.Parent or not activeSpectatingModel:FindFirstChild("Humanoid") or activeSpectatingModel.Humanoid.Health <= 0 then
            local foundNew = nil
            for _, obj in ipairs(workspace:GetDescendants()) do
                if obj:IsA("Model") and obj.Name == activeSpectatingModel.Name and obj:FindFirstChild("Humanoid") and obj.Humanoid.Health > 0 then
                    local isPlayerChar = false
                    for _, p in ipairs(Players:GetPlayers()) do
                        if p.Character == obj then isPlayerChar = true break end
                    end
                    if not isPlayerChar then
                        foundNew = obj
                        break
                    end
                end
            end
            if foundNew then
                activeSpectatingModel = foundNew
            end
        end

        if activeSpectatingModel and activeSpectatingModel:FindFirstChild("Humanoid") then
            Camera.CameraSubject = activeSpectatingModel.Humanoid
            local hrp = activeSpectatingModel:FindFirstChild("HumanoidRootPart")
            if hrp then
                pcall(function() LocalPlayer:RequestStreamAroundAsync(hrp.Position) end)
            end
        else
            local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if hum then Camera.CameraSubject = hum end
        end
    else
        local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if hum and Camera.CameraSubject ~= hum then
            Camera.CameraSubject = hum
        end
    end

    -- Fly Movement
    if flyEnabled and flyBV and flyBG then
        local moveDir = Vector3.zero
        local camCF = Camera.CFrame
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir += camCF.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir -= camCF.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir -= camCF.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir += camCF.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDir += Vector3.new(0, 1, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) or UserInputService:IsKeyDown(Enum.KeyCode.C) then
            moveDir -= Vector3.new(0, 1, 0)
        end
        flyBG.CFrame = camCF
        flyBV.Velocity = (moveDir.Magnitude > 0 and moveDir.Unit or Vector3.zero) * flySpeed
    end

    -- Godmode & WalkSpeed enforcement
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then
        if godmodeEnabled and hum.Health < hum.MaxHealth then
            hum.Health = hum.MaxHealth
        end
        if targetWalkSpeed ~= 16 and hum.WalkSpeed ~= targetWalkSpeed then
            hum.WalkSpeed = targetWalkSpeed
        end
    end

    -- Live ESP Auto-Scan for Bots/NPCs & Coordinate Updates (from original script)
    if scanningEnabled then
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("Model") and obj:FindFirstChild("Humanoid") and obj:FindFirstChild("HumanoidRootPart") then
                local isPlayerChar = false
                for _, p in ipairs(Players:GetPlayers()) do
                    if p.Character == obj then isPlayerChar = true break end
                end
                if not isPlayerChar and not espCache[obj] then
                    addESP(obj, obj.Name)
                end
            end
        end

        local myHrp = char and char:FindFirstChild("HumanoidRootPart")
        for targetChar, data in pairs(espCache) do
            if data.rootPart and data.textLabel and data.rootPart.Parent then
                local pos = data.rootPart.Position
                local dist = myHrp and math.floor((myHrp.Position - pos).Magnitude) or 0
                data.textLabel.Text = string.format("[%s] %dm\nPos: X:%.0f, Y:%.0f, Z:%.0f", data.name, dist, pos.X, pos.Y, pos.Z)
            else
                removeESP(targetChar)
            end
        end
    end
end)

-- Player Join / Leave Hooks (including Client-View Hidden persistence)
local function hookPlayer(player)
    player.CharacterAdded:Connect(function(char)
        if clientHiddenUsers[player.UserId] then
            task.wait(0.2)
            setCharacterHiddenLocal(char, true)
        elseif scanningEnabled then
            task.wait(1)
            addESP(char, player.Name)
        end
    end)
end

for _, p in ipairs(Players:GetPlayers()) do
    if p ~= LocalPlayer then hookPlayer(p) end
end

Players.PlayerAdded:Connect(function(player)
    hookPlayer(player)
    if updateSpectateList and not isBotMode then updateSpectateList() end
    if refreshTargetSelector then refreshTargetSelector() end
end)

Players.PlayerRemoving:Connect(function(player)
    if activeSpectatingPlayer == player then
        activeSpectatingPlayer = nil
        setStatus("DELTA ADMIN · " .. T("subtitle"))
    end
    if jailModels[player.UserId] then
        jailModels[player.UserId]:Destroy()
        jailModels[player.UserId] = nil
    end
    if player.Character then removeESP(player.Character) end
    if updateSpectateList and not isBotMode then updateSpectateList() end
    if refreshTargetSelector then refreshTargetSelector() end
end)

-- Bottom Reset View (Click = Reset Camera, Hold 2s = Toggle Bot Mode like original script)
local pressStartTime = 0
local isHolding = false

ResetButton.MouseButton1Down:Connect(function()
    pressStartTime = tick()
    isHolding = true
    task.spawn(function()
        while isHolding do
            if tick() - pressStartTime >= 2 then
                isBotMode = not isBotMode
                activeSpectatingPlayer = nil
                activeSpectatingModel = nil
                ResetButton.TextColor3 = isBotMode and Color3.fromRGB(255, 50, 50) or TEXT_MAIN
                setStatus(isBotMode and "BOT SPECTATE" or "PLAYER SPECTATE")
                if updateSpectateList then updateSpectateList() end
                isHolding = false
                break
            end
            task.wait(0.1)
        end
    end)
end)

ResetButton.MouseButton1Up:Connect(function()
    if isHolding then
        isHolding = false
        if tick() - pressStartTime < 2 then
            activeSpectatingPlayer = nil
            activeSpectatingModel = nil
            setStatus("DELTA ADMIN · " .. T("subtitle"))
            if updateSpectateList then updateSpectateList() end
        end
    end
end)
